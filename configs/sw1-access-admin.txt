! ============================================================
! Device : SW1-Access-Admin
! Model  : Cisco 2960-24TT
! Role   : Access switch, Administration building
! Connects to:
!   G0/1       MLS1-Core G1/0/3 (trunk)
!   G0/2       MLS2-Core G1/0/3 (trunk)
!   Fa0/1-8    Staff PCs (VLAN 20)
!   Fa0/9-12   IoT: door lock, smoke detector, alarm (VLAN 50)
!   Fa0/13     Admin PC (VLAN 99)
!   Fa0/14-24  Unused (shut, VLAN 998)
! Addresses: docs/addressing.md   Explanations: docs/build-guide.md
! ============================================================
!
! Lines starting with "!" are comments; IOS ignores them when pasted.
! Paste one "--- 1.x" section at a time and watch for "% Invalid input".
!
! ==================== STAGE 1: Base + Layer 2 ====================
!
enable
configure terminal
!
! --- 1.1 Identity ---
hostname SW1-Access-Admin
no ip domain-lookup
!
! --- 1.2 Passwords and local account ---
enable secret RVU@enable1
service password-encryption
username netadmin privilege 15 secret RVU@admin1
banner motd # Authorized access only. RVU Campus Network. All activity is monitored. #
!
! --- 1.3 SSH ---
ip domain-name rvu-campus.local
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
! --- 1.4 VLANs (only the ones used in this building) ---
vtp mode transparent
vlan 20
 name FACULTY
vlan 50
 name IOT
vlan 99
 name MANAGEMENT
vlan 998
 name BLACKHOLE
vlan 999
 name NATIVE-UNUSED
exit
!
! --- 1.6 Trunk uplinks to both core switches ---
interface g0/1
 description Trunk to MLS1-Core G1/0/3
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,50,99
 switchport nonegotiate
 no shutdown
interface g0/2
 description Trunk to MLS2-Core G1/0/3
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,50,99
 switchport nonegotiate
 no shutdown
exit
!
! --- 1.7 Access ports ---
interface range fa0/1 - 8
 description Staff PCs
 switchport mode access
 switchport access vlan 20
 no shutdown
interface range fa0/9 - 12
 description IoT devices
 switchport mode access
 switchport access vlan 50
 no shutdown
interface fa0/13
 description Admin PC
 switchport mode access
 switchport access vlan 99
 no shutdown
interface range fa0/14 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
exit
!
! --- 1.8 Management IP ---
! The gateway (HSRP virtual IP) only answers after Stage 5. SSH from the
! admin PC works before that because both are in the same subnet.
interface vlan 99
 description Management
 ip address 10.10.3.101 255.255.255.224
 no shutdown
exit
ip default-gateway 10.10.3.97
!
end
write memory
