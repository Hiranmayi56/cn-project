! ============================================================
! Device : SW-DMZ
! Model  : Cisco 2960-24TT
! Role   : DMZ switch for public-facing servers. Not connected to the core:
!          its only way out is R1-Edge, so it has no trunks.
! Connects to:
!   G0/1       R1-Edge G0/2 (VLAN 60, R1-Edge is the gateway 10.10.3.129)
!   Fa0/1      Web server   10.10.3.130 (VLAN 60)
!   Fa0/2      DMZ DNS      10.10.3.131 (VLAN 60)
!   Fa0/3      Email server 10.10.3.132 (VLAN 60)
!   Fa0/4-24, G0/2  Unused (shut, VLAN 998)
! Addresses: docs/addressing.md   Explanations: docs/build-guide.md
! ============================================================
!
! Lines starting with "!" are comments; IOS ignores them when pasted.
! Paste one "--- 1.x" section at a time and watch for "% Invalid input".
!
! ==================== STAGE 1: Base + Layer 2 ====================
!
enable
configure terminal
!
! --- 1.1 Identity ---
hostname SW-DMZ
no ip domain-lookup
!
! --- 1.2 Passwords and local account ---
enable secret RVU@enable1
service password-encryption
username netadmin privilege 15 secret RVU@admin1
banner motd # Authorized access only. RVU Campus Network. All activity is monitored. #
!
! --- 1.3 SSH ---
ip domain-name rvu-campus.local
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
! --- 1.4 VLANs (no trunks here, so no native VLAN is needed) ---
vtp mode transparent
vlan 60
 name DMZ
vlan 998
 name BLACKHOLE
exit
!
! --- 1.7 Access ports ---
interface g0/1
 description Uplink to R1-Edge G0/2
 switchport mode access
 switchport access vlan 60
 no shutdown
interface fa0/1
 description Web server
 switchport mode access
 switchport access vlan 60
 no shutdown
interface fa0/2
 description DMZ DNS server
 switchport mode access
 switchport access vlan 60
 no shutdown
interface fa0/3
 description Email server
 switchport mode access
 switchport access vlan 60
 no shutdown
interface range fa0/4 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
interface g0/2
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
exit
!
! --- 1.8 Management IP ---
! SSH to this switch only works once R1-Edge routes to the DMZ (Stage 6).
! Until then, configure and check it from the console (CLI tab).
interface vlan 60
 description Management (DMZ)
 ip address 10.10.3.140 255.255.255.240
 no shutdown
exit
ip default-gateway 10.10.3.129
!
end
write memory
