! ============================================================
! Device : SW-Servers
! Model  : Cisco 2960-24TT
! Role   : Server block switch (server room, Administration building)
! Connects to:
!   G0/1       MLS1-Core G1/0/6 (trunk)
!   G0/2       MLS2-Core G1/0/6 (trunk)
!   Fa0/1      DHCP server         10.10.3.70 (VLAN 30)
!   Fa0/2      Internal DNS server 10.10.3.71 (VLAN 30)
!   Fa0/3      IoT server          10.10.3.72 (VLAN 30)
!   Fa0/4-24   Unused (shut, VLAN 998)
! Addresses: docs/addressing.md   Explanations: docs/build-guide.md
! ============================================================
!
! Lines starting with "!" are comments; IOS ignores them when pasted.
! Paste one "--- 1.x" section at a time and watch for "% Invalid input".
!
! ==================== STAGE 1: Base + Layer 2 ====================
!
enable
configure terminal
!
! --- 1.1 Identity ---
hostname SW-Servers
no ip domain-lookup
!
! --- 1.2 Passwords and local account ---
enable secret RVU@enable1
service password-encryption
username netadmin privilege 15 secret RVU@admin1
banner motd # Authorized access only. RVU Campus Network. All activity is monitored. #
!
! --- 1.3 SSH ---
ip domain-name rvu-campus.local
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
! --- 1.4 VLANs ---
vtp mode transparent
vlan 30
 name SERVERS
vlan 99
 name MANAGEMENT
vlan 998
 name BLACKHOLE
vlan 999
 name NATIVE-UNUSED
exit
!
! --- 1.6 Trunk uplinks to both core switches ---
interface g0/1
 description Trunk to MLS1-Core G1/0/6
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99
 switchport nonegotiate
 no shutdown
interface g0/2
 description Trunk to MLS2-Core G1/0/6
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99
 switchport nonegotiate
 no shutdown
exit
!
! --- 1.7 Access ports ---
interface fa0/1
 description DHCP server
 switchport mode access
 switchport access vlan 30
 no shutdown
interface fa0/2
 description Internal DNS server
 switchport mode access
 switchport access vlan 30
 no shutdown
interface fa0/3
 description IoT server
 switchport mode access
 switchport access vlan 30
 no shutdown
interface range fa0/4 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
exit
!
! --- 1.8 Management IP ---
interface vlan 99
 description Management
 ip address 10.10.3.104 255.255.255.224
 no shutdown
exit
ip default-gateway 10.10.3.97
!
end
write memory
