! ============================================================
! Device : MLS1-Core
! Model  : Cisco 3650-24PS (needs a power supply: Physical tab -> AC-POWER-SUPPLY)
! Role   : Core/distribution switch 1. Inter-VLAN routing, HSRP and DHCP relay
!          are added in later stages. HSRP active + STP root for VLANs 10, 30, 50.
! Connects to:
!   G1/0/1-2  EtherChannel Po1 (LACP) to MLS2-Core G1/0/1-2
!   G1/0/3    SW1-Access-Admin G0/1   (trunk)
!   G1/0/4    SW2-Access-Acad  G0/1   (trunk)
!   G1/0/5    SW3-Access-Lib   G0/1   (trunk)
!   G1/0/6    SW-Servers       G0/1   (trunk)
!   G1/0/24   R1-Edge G0/0            (routed link, configured in Stage 3)
! Addresses: docs/addressing.md   Explanations: docs/build-guide.md
! ============================================================
!
! Lines starting with "!" are comments; IOS ignores them when pasted.
! Paste one "--- 1.x" section at a time and watch for "% Invalid input".
!
! ==================== STAGE 1: Base + Layer 2 ====================
!
enable
configure terminal
!
! --- 1.1 Identity ---
hostname MLS1-Core
no ip domain-lookup
!
! --- 1.2 Passwords and local account ---
enable secret RVU@enable1
service password-encryption
username netadmin privilege 15 secret RVU@admin1
banner motd # Authorized access only. RVU Campus Network. All activity is monitored. #
!
! --- 1.3 SSH ---
ip domain-name rvu-campus.local
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 login local
 logging synchronous
 exec-timeout 15 0
line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit
!
! --- 1.4 VLANs ---
vtp mode transparent
vlan 10
 name STUDENTS
vlan 20
 name FACULTY
vlan 30
 name SERVERS
vlan 40
 name GUEST
vlan 50
 name IOT
vlan 99
 name MANAGEMENT
vlan 998
 name BLACKHOLE
vlan 999
 name NATIVE-UNUSED
exit
!
! --- 1.5 EtherChannel Po1 to MLS2-Core ---
! If "switchport trunk encapsulation dot1q" gives "% Invalid input", ignore it:
! that switch only supports 802.1Q, so the line isn't needed.
interface range g1/0/1 - 2
 shutdown
 description EtherChannel Po1 to MLS2-Core
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,99
 switchport nonegotiate
 channel-group 1 mode active
exit
interface port-channel 1
 description Po1 to MLS2-Core
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,50,99
 switchport nonegotiate
exit
interface range g1/0/1 - 2
 no shutdown
exit
!
! --- 1.6 Trunks to access and server switches ---
interface g1/0/3
 description Trunk to SW1-Access-Admin G0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,50,99
 switchport nonegotiate
 no shutdown
interface g1/0/4
 description Trunk to SW2-Access-Acad G0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,50,99
 switchport nonegotiate
 no shutdown
interface g1/0/5
 description Trunk to SW3-Access-Lib G0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,40,50,99
 switchport nonegotiate
 no shutdown
interface g1/0/6
 description Trunk to SW-Servers G0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99
 switchport nonegotiate
 no shutdown
exit
!
! --- 1.7 Unused ports (G1/0/24 is left alone until Stage 3) ---
interface range g1/0/7 - 23
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
exit
! Only if your 3650 shows G1/1/1-4 uplink ports (skip on "% Invalid input"):
interface range g1/1/1 - 4
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
exit
!
! --- 1.8 Management IP ---
interface vlan 99
 description Management
 ip address 10.10.3.98 255.255.255.224
 no shutdown
exit
!
end
write memory
